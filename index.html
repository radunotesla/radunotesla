// --- LOGICA INTEGRATA PER PERCORSO STRADALE REALE INLINE --- let mapInstance = null; let routingControl = null; document.getElementById('btn-calcola').addEventListener('click', function() { const latVal = document.getElementById('cittaLat').value; const lonVal = document.getElementById('cittaLon').value; const campoDistanza = document.getElementById('distanza'); const boxMappa = document.getElementById('contenitore-mappa'); if (!latVal || !lonVal) { alert("Per favore, digita e seleziona prima una città valida dall'elenco dei suggerimenti."); return; } const latPartenza = parseFloat(latVal); const lonPartenza = parseFloat(lonVal); // Forza l'apertura visiva immediata del contenitore sulla pagina boxMappa.style.height = "320px"; boxMappa.style.marginTop = "15px"; boxMappa.style.border = "1px solid #ccc"; // Inizializzazione della mappa Leaflet if (!mapInstance) { mapInstance = L.map('contenitore-mappa').setView([LAT_RADUNO, LON_RADUNO], 8); L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { attribution: '© OpenStreetMap' }).addTo(mapInstance); } else { // Se l'utente ricalcola una nuova città, rimuove la rotta precedente per non sovrapporla if (routingControl) { mapInstance.removeControl(routingControl); } } // Risolve il bug del riquadro vuoto/grigio forzando il calcolo dei pixel a schermo aperto setTimeout(function() { mapInstance.invalidateSize(true); // Richiesta stradale reale al server OSRM routingControl = L.Routing.control({ waypoints: [ L.latLng(latPartenza, lonPartenza), L.latLng(LAT_RADUNO, LON_RADUNO) ], router: L.Routing.osrmv1({ serviceUrl: 'https://project-osrm.org' }), lineOptions: { styles: [{ color: '#cc0000', weight: 6, opacity: 0.85 }] // Linea rossa Tesla }, createMarker: function() { return null; } // Nasconde i pin predefiniti di Leaflet }).addTo(mapInstance); // Quando la rotta stradale viene trovata con successo routingControl.on('routesfound', function(e) { const routes = e.routes; const distanzaMetri = routes.summary.totalDistance; const kmStradaliVeri = Math.round(distanzaMetri / 1000); // Converte i metri in KM effettivi // Scrive stabilmente il chilometraggio stradale nel campo del modulo if (campoDistanza) { campoDistanza.value = kmStradaliVeri + " KM"; } // Zooma e centra automaticamente la mappa per inquadrare l'intero viaggio const bounds = L.latLngBounds(routes.coordinates); mapInstance.fitBounds(bounds); }); }, 200); // Micro-ritardo di 200ms per garantire la corretta visualizzazione delle mappe });
